{{/* Work type filter — text tabs above the project grid on md+, a select on small screens where four labels would not fit on one line. Call with a dict: "active" the filter key currently shown ("" for All) "alpine" true on /work/, where the controls also drive the x-data scope declared in layouts/work/list.html. Otherwise (the per-type pages, or no JS) the tabs are plain links to /work// and the select navigates there. Types come from each entry's `site_type`; an entry with none counts as "new", matching the badge in project-card.html. */}} {{- $active := .active | default "" -}} {{- $alpine := .alpine | default false -}} {{- $all := (site.GetPage "/work").RegularPages -}} {{- $types := slice (dict "key" "" "label" "All" "n" (len $all)) (dict "key" "new" "label" "New sites" "n" (len (where $all "Params.site_type" "not in" (slice "rescue" "open-source")))) (dict "key" "rescue" "label" "Rescues" "n" (len (where $all "Params.site_type" "eq" "rescue"))) (dict "key" "open-source" "label" "Open source" "n" (len (where $all "Params.site_type" "eq" "open-source"))) -}} {{- $on := "text-cream border-ember" -}} {{- $off := "text-cream/50 border-transparent hover:text-cream" -}}
{{ range $types }} {{- if .n }} {{- $key := .key -}} {{ .label }} {{ .n }} {{- end }} {{ end }}
